6. Transformaciones de un modelo (E04)

Esta aplicación introduce la posibilidad de mover el objeto, es decir, realizar transformaciones de rotación, traslación y escalado.

Hereda las acciones de E03. La iteración con el usuario es mediante teclado. Así con 'F1' se realizan rotaciones, 'F2' traslaciones y 'F3' escalados. De esta forma, pulsando 'F1' y moviendo los cursores el objeto rotará con relación a los ejes de coordenadas X e Y, no permitiéndose la rotación en el eje Z. Se añade el parámetro que activa el doble buffer (GLUT_DOUBLE), al llamar a la función glutInitDisplayMode.

Funciones añadidas en esta aplicación:

  • glutSwapBuffers() - Intercambia los buffers, ya no es necesario llamar a la función glFlush(), pues antes de realizar el intercambio OpenGL completa todas las tareas de dibujado que están pendientes.

  • glLoadMatrixf( glMatrix ) - Reemplaza la matriz actual por la indicada en el parámetro de la función.

  • glPopMatrix() - Saca la matriz actual de la pila de matrices.

  • glPushMatrix() - Coloca la matriz actual en la pila de matrices. Se utiliza para guardar el estado actual, de manera que pueda ser recuperada llamando a glPopMatrix().

  • glutSpecialFunc( SpecialKey ) - Realiza la misma función que glutKeyboardFunc, solo que en este caso controla las teclas especiales, como los Fx, cursores, etc

  • glutCreateMenu( MainMenu ) - Añade un menú flotante. Su único argumento es la función a la que se llama cuando se activa el menú.

  • glutAddMenuEntry( "Salir", 0 ) - Añade una entrada al menú, los parámetros indican la etiqueta y el parámetro dentro de "MainMenu".

  • glutAttachMenu( GLUT_RIGHT_BUTTON ) - Indica cuando se activa el menú, en este caso cuando se pulsa el botón derecho del ratón.

Aparte de las funciones aquí indicadas propias de OpenGL se utiliza el fichero Trfm3D, en el que se encuentra la estructura matriz3D.

Ver en el código fuente las declaraciones de esas estructuras y de las funciones que tienen asociadas.

#include < windows.h >
#include < GL/gl.h >
#include < GL/glut.h >

#include < list.h >
#include < polygon.h >
#include < wavefront.h >
#include < trfm3D.h >

#define WIDTH  200		/* Ancho ventanas en pixels */
#define HEIGHT 200		/* Alto ventanas en pixels */


#define ROTATE		101
#define MOVE		102
#define SCALE		103


// state data
static GLfloat  FEA_Red    = 1.0f;
static GLfloat  FEA_Green  = 1.0f;
static GLfloat  FEA_Blue   = 1.0f;
static int      FEA_Action = ROTATE;

// model data
static trfm3D  *FEA_Transform;
static list    *FEA_ListPol;


void Display( void )
{
    polygon *poly;
	int      j = 1;
	int      i;
	int      numPol;
    GLfloat  glMatrix[16];


	// we asume we are in GL_MODELVIEW state
	// and the current matrix is the unit matrix

	glPushMatrix();

	SetGlMatrix( FEA_Transform, glMatrix );
	glLoadMatrixf( glMatrix );
	
	glClear( GL_COLOR_BUFFER_BIT );
	glColor3f( 1-FEA_Red, 1-FEA_Green, 1-FEA_Blue );

	poly = StartLoop( FEA_ListPol );

	numPol = ElementsInList( FEA_ListPol );

	while ( j <= numPol )
	{ 
		for ( i=0; i < poly->numVert; i++ )
		{
			glBegin( GL_LINES );
				glVertex3f( poly->xCoor[i], poly->yCoor[i], 
poly->zCoor[i] );
glVertex3f( poly->xCoor[i+1], poly->yCoor[i+1], 
poly->zCoor[i+1] );
			glEnd();
		}
		j++;
		poly=GetNext( FEA_ListPol );
	}

	glPopMatrix();
	glutSwapBuffers();
}


void Resize( int w, int h )
{
	glClearColor(1.0, 1.0, 1.0, 1.0);

	glViewport( 0, 0, w, h );

	glMatrixMode( GL_PROJECTION );
	glLoadIdentity();
	glOrtho( -50.0, 50.0, -50.0, 50.0, -50.0, 50.0 );

	glMatrixMode( GL_MODELVIEW );
	glLoadIdentity();


}

void Keyboard ( unsigned char key, int x, int y)
{
	switch( key )
	{
	case 'r': 
		if ( FEA_Red != 0.0 ) FEA_Red = FEA_Red - 0.1;
		
		break;
	case 'R':
		if ( FEA_Red != 1.0 ) FEA_Red = FEA_Red + 0.1;
		break;
	case 'b': 
		if ( FEA_Blue != 0.0 ) FEA_Blue = FEA_Blue - 0.1;
		break;
	case 'B':
		if ( FEA_Blue != 1.0 ) FEA_Blue = FEA_Blue + 0.1;
		break;
	case 'g': 
		if ( FEA_Green != 0.0 ) FEA_Green = FEA_Green - 0.1;
		break;
	case 'G':
		if ( FEA_Green != 1.0 ) FEA_Green = FEA_Green + 0.1;
		break;
	case 27: // 
		exit(0);
		break;
	}
	glutPostRedisplay();
}


static void SpecialKey(int k, int x, int y) 
{
	switch(k) 
	{
	case GLUT_KEY_UP:
		{
		if      (FEA_Action == ROTATE) 
AddRotX  ( FEA_Transform, 0.5f );
		else if (FEA_Action == MOVE )  
AddTrans ( FEA_Transform, 0.0f, 5.0f, 0.0f );
		else if (FEA_Action == SCALE ) 
AddScale ( FEA_Transform, 0.9f );
		}
		break;
	case GLUT_KEY_DOWN:
		{
		if      (FEA_Action == ROTATE) 
AddRotX  ( FEA_Transform, -0.5 );
		else if (FEA_Action == MOVE )  
AddTrans ( FEA_Transform, 0.0, -5.0, 0.0 );
		else if (FEA_Action == SCALE ) 
AddScale ( FEA_Transform, 1.1f );
		}
		break;
	case GLUT_KEY_RIGHT:
		{
		if      (FEA_Action == ROTATE) 
AddRotY  ( FEA_Transform, 0.5 );
		else if (FEA_Action == MOVE )  
AddTrans ( FEA_Transform, 5.0, 0.0, 0.0 );
		}
		break;
	case GLUT_KEY_LEFT:
		{
		if      (FEA_Action == ROTATE) 
AddRotY  ( FEA_Transform, -0.5 );
		else if (FEA_Action == MOVE )  
AddTrans ( FEA_Transform, -5.0, 0.0, 0.0 );
		}
		break;
	case GLUT_KEY_F1:
		FEA_Action = ROTATE;
		break;
	case GLUT_KEY_F2:
		FEA_Action = MOVE;
		break;
	case GLUT_KEY_F3:
		FEA_Action = SCALE;
		break;
	case GLUT_KEY_F4:
		DestroyTrfm3D( FEA_Transform );
		FEA_Transform = CreateTrfm3D();
		break;	
	default:
		break;
	}
	glutPostRedisplay();
}

#ifndef TRFM3D_H
#define TRFM3D_H

typedef struct _Trfm3D
{
    /* la matriz la almacenamos descompuesta en elementos */
    float  r1X, r1Y, r1Z;
    float  r2X, r2Y, r2Z;
    float  r3X, r3Y, r3Z;
    float  scl;
    float  trX, trY, trZ;
} trfm3D;

    /* aloca memoria para una estructura trfm3D, 
       da valor a los elementos de la estructura,
       instala la transformacion identidad y devuelve el pointer */
void    DestroyTrfm3D( trfm3D * );
trfm3D *CreateTrfm3D();
trfm3D *CopyTrfm3D( trfm3D * );

void SetGlMatrix( trfm3D *thisTrfm3D, GLfloat *glmatrix );

    /* El pointer que se pasa debe apuntar a una estructura creada previamente
       Los angulos se pasan en radianes (ver comentario abajo) */
void RotX(  trfm3D * , float angle );
void RotY(  trfm3D * , float angle );
void RotZ(  trfm3D * , float angle );
void Trans( trfm3D * , float tx, float ty, float tz );
void Scale( trfm3D * , float scale );

void AddRotX(  trfm3D * , float angle );

void AddRotZ(  trfm3D * , float angle );
void AddTrans( trfm3D * , float tx, float ty, float tz );

    /* compone la transformacion add sobre la transformacion previous
       [previous] = [previous] * [add] */
void ComponerTrfm3D( trfm3D *previous, trfm3D *add );

#endif /* TRFM3D_H */


Trfm3D.c

 
#include < windows.h >
#include < GL/gl.h >
#include < stdlib.h >
#include < math.h >

#include < gc.h >
#include < trfm3D.h >

void DestroyTrfm3D( trfm3D *thisTrfm3D )
{
    free( thisTrfm3D );
}

trfm3D *CreateTrfm3D()
{
    trfm3D *unitTrfm;

    unitTrfm = (trfm3D *) malloc( sizeof(trfm3D) * 1 );

    unitTrfm->r1X = 1.0;
    unitTrfm->r1Y = 0.0;
    unitTrfm->r1Z = 0.0;
    unitTrfm->r2X = 0.0;
    unitTrfm->r2Y = 1.0;
    unitTrfm->r2Z = 0.0;
    unitTrfm->r3X = 0.0;
    unitTrfm->r3Y = 0.0;
    unitTrfm->r3Z = 1.0;
    unitTrfm->scl = 1.0;
    unitTrfm->trX = 0.0;
    unitTrfm->trY = 0.0;
    unitTrfm->trZ = 0.0;

    return unitTrfm;
}

void SetGlMatrix( trfm3D *thisTrfm3D, GLfloat *glmatrix )
{
	glmatrix[0]  = thisTrfm3D->r1X;
	glmatrix[1]  = thisTrfm3D->r1Y;
	glmatrix[2]  = thisTrfm3D->r1Z;
	glmatrix[3]  = 0.0;
	glmatrix[4]  = thisTrfm3D->r2X;
	glmatrix[5]  = thisTrfm3D->r2Y;
	glmatrix[6]  = thisTrfm3D->r2Z;
	glmatrix[7]  = 0.0;
	glmatrix[8]  = thisTrfm3D->r3X;
	glmatrix[9]  = thisTrfm3D->r3Y;
	glmatrix[10] = thisTrfm3D->r3Z;
	glmatrix[11]  = 0.0;
	glmatrix[12] = thisTrfm3D->trX;
	glmatrix[13] = thisTrfm3D->trY;
	glmatrix[14] = thisTrfm3D->trZ;
	glmatrix[15] = thisTrfm3D->scl;
  }

void RotX( trfm3D *thisTrfm3D, float angle )
{
    thisTrfm3D->r1X = 1.0;
    thisTrfm3D->r1Y = 0.0;
    thisTrfm3D->r1Z = 0.0;
    thisTrfm3D->r2X = 0.0;
    thisTrfm3D->r2Y = (float)   cos( (double) angle );
    thisTrfm3D->r2Z = (float)   sin( (double) angle );
    thisTrfm3D->r3X = 0.0;
    thisTrfm3D->r3Y = (float) - sin( (double) angle );
    thisTrfm3D->r3Z = (float)   cos( (double) angle );
    thisTrfm3D->trX = 0.0;
    thisTrfm3D->trY = 0.0;
    thisTrfm3D->trZ = 0.0;
}

void RotY( trfm3D *thisTrfm3D, float angle )
{
    thisTrfm3D->r1X = (float)   cos( (double) angle );
    thisTrfm3D->r1Y = 0.0;
    thisTrfm3D->r1Z = (float) - sin( (double) angle );
    thisTrfm3D->r2X = 0.0;
    thisTrfm3D->r2Y = 1.0;
    thisTrfm3D->r2Z = 0.0;
    thisTrfm3D->r3X = (float)   sin( (double) angle );
    thisTrfm3D->r3Y = 0.0;
    thisTrfm3D->r3Z = (float)   cos( (double) angle );
    thisTrfm3D->trX = 0.0;
    thisTrfm3D->trY = 0.0;
    thisTrfm3D->trZ = 0.0;
}

void RotZ( trfm3D *thisTrfm3D, float angle )
{
    thisTrfm3D->r1X = (float)   cos( (double) angle );
    thisTrfm3D->r1Y = (float)   sin( (double) angle );
    thisTrfm3D->r1Z = 0.0;
    thisTrfm3D->r2X = (float) - sin( (double) angle );
    thisTrfm3D->r2Y = (float)   cos( (double) angle );
    thisTrfm3D->r2Z = 0.0;
    thisTrfm3D->r3X = 0.0;
    thisTrfm3D->r3Y = 0.0;
    thisTrfm3D->r3Z = 1.0;
    thisTrfm3D->trX = 0.0;
    thisTrfm3D->trY = 0.0;
    thisTrfm3D->trZ = 0.0;
}

void Trans( trfm3D *thisTrfm3D, float tx, float ty, float tz )
{
    thisTrfm3D->r1X = 1.0;
    thisTrfm3D->r1Y = 0.0;
    thisTrfm3D->r1Z = 0.0;
    thisTrfm3D->r2X = 0.0;
    thisTrfm3D->r2Y = 1.0;
    thisTrfm3D->r2Z = 0.0;
    thisTrfm3D->r3X = 0.0;
    thisTrfm3D->r3Y = 0.0;
    thisTrfm3D->r3Z = 1.0;
    thisTrfm3D->trX = tx;
    thisTrfm3D->trY = ty;
    thisTrfm3D->trZ = tz;
}

void Scale( trfm3D *thisTrfm3D, float scale )
{
    thisTrfm3D->r1X = 1.0;
    thisTrfm3D->r1Y = 0.0;
    thisTrfm3D->r1Z = 0.0;
    thisTrfm3D->r2X = 0.0;
    thisTrfm3D->r2Y = 1.0;
    thisTrfm3D->r2Z = 0.0;
    thisTrfm3D->r3X = 0.0;
    thisTrfm3D->r3Y = 0.0;
    thisTrfm3D->r3Z = 1.0;
    thisTrfm3D->scl = scale;
    thisTrfm3D->trX = 0.0;
    thisTrfm3D->trY = 0.0;
    thisTrfm3D->trZ = 0.0;
}

void AddRotX(  trfm3D *ThisMatrix , float angle )
{
    float   addR2Y, addR2Z;
    float   addR3Y, addR3Z;
    float   prvR1X, prvR1Y, prvR1Z;
    float   prvR2X, prvR2Y, prvR2Z;
    float   prvR3X, prvR3Y, prvR3Z;
    float   prvTrX, prvTrY, prvTrZ;
    float   prvScl;
    float   newR1X, newR1Y, newR1Z;
    float   newR2X, newR2Y, newR2Z;
    float   newR3X, newR3Y, newR3Z;
    float   mod;

    prvScl = ThisMatrix->scl;
    prvR1X = ThisMatrix->r1X * prvScl;
    prvR1Y = ThisMatrix->r1Y * prvScl;
    prvR1Z = ThisMatrix->r1Z * prvScl;
    prvR2X = ThisMatrix->r2X * prvScl;
    prvR2Y = ThisMatrix->r2Y * prvScl;
    prvR2Z = ThisMatrix->r2Z * prvScl;
    prvR3X = ThisMatrix->r3X * prvScl;
    prvR3Y = ThisMatrix->r3Y * prvScl;
    prvR3Z = ThisMatrix->r3Z * prvScl;
    prvTrX = ThisMatrix->trX;
    prvTrY = ThisMatrix->trY;
    prvTrZ = ThisMatrix->trZ;

    addR2Y = (float)   cos((double) angle );
    addR2Z = (float)   sin((double) angle );
    addR3Y = (float) - sin((double) angle );
    addR3Z = (float)   cos((double) angle );

    newR1X = prvR1X;
    newR1Y = prvR1Y * addR2Y + prvR1Z * addR3Y;
    newR1Z = prvR1Y * addR2Z + prvR1Z * addR3Z;
    newR2X = prvR2X;
    newR2Y = prvR2Y * addR2Y + prvR2Z * addR3Y;
    newR2Z = prvR2Y * addR2Z + prvR2Z * addR3Z;
    newR3X = prvR3X;
    newR3Y = prvR3Y * addR2Y + prvR3Z * addR3Y;
    newR3Z = prvR3Y * addR2Z + prvR3Z * addR3Z;
    ThisMatrix->trX = prvTrX;
    ThisMatrix->trY = prvTrY * addR2Y + prvTrZ * addR3Y;
    ThisMatrix->trZ = prvTrY * addR2Z + prvTrZ * addR3Z;

    mod = newR1X * newR1X + newR2X * newR2X + newR3X * newR3X;
    mod = (float) sqrt( mod );
    ThisMatrix->scl = mod;
    mod = (float) (1. / mod);
    
	ThisMatrix->r1X = newR1X * (float) mod;
    ThisMatrix->r1Y = newR1Y * (float) mod;
    ThisMatrix->r1Z = newR1Z * (float) mod;
    ThisMatrix->r2X = newR2X * (float) mod;
    ThisMatrix->r2Y = newR2Y * (float) mod;
    ThisMatrix->r2Z = newR2Z * (float) mod;
    ThisMatrix->r3X = newR3X * (float) mod;
    ThisMatrix->r3Y = newR3Y * (float) mod;
    ThisMatrix->r3Z = newR3Z * (float) mod;
}

void AddRotZ(  trfm3D *ThisMatrix , float angle )
