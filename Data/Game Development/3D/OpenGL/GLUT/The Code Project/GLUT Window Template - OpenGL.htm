Introduction

This program sets up an OpenGL window on any platform using the GLUT library. GLUT stands for GL Utility Toolkit. This program just displays a window on the screen. The background color of the window is set to black. You can use this code as a template for your OpenGL programs so that you never have to write the same code every time you want to set up an OpenGL window. The code is fully commented...

Using the code

1     //*************************************************************************
2     //
3     //  File Name: Glut Window Template
4     //  Author   : Ali BaderEddin
5     //  
6     //  Description: Just opening a window using GLUT library...
7     //  
8     //*************************************************************************

9
10    #include <gl/glut.h>
11
12    // Link the lib files to the program. This is not necessary 
13    // if you have added the lib names using Project/Settings/Link
14    #pragma comment (lib, "opengl32.lib")
15    #pragma comment (lib, "glut32.lib")
16    #pragma comment (lib, "glu32.lib")
17
18    //  define the window position on screen
19    int window_X_position;
20    int window_Y_position;
21
22    //  variables representing the window size
23    int window_width = 512;
24    int window_height = 512;
25
26    //  variable representing the window title
27    char *window_title = "Lesson 1";
28
29    //  Tells whether to display the window full screen or not
30    //  This is a boolean, but you know C doesn't support boolean.
31    //  Press Alt + Esc to exit a full screen.
32    int fullScreen = 0;
33
34    //  Function that initializes some values that won't change
35    //  throughout the program...
36    void init (void);
37
38    //  This function is used to display all the graphics objects
39    //  on the screen. It is the most important function in OpenGL.
40    //  Don't be confused, you can give 'display' another name like 'foo'...
41    //  This function gains its importance when you write the command
42    //  glutDisplayFunc (display) or glutDisplayFunc (foo) in function 'main'.
43    void display (void);
44
45    //  This function centers your window on the screen
46    void centerOnScreen (void);
47
48    void init ()
49       {
50        //  Set the clear color to be black. If you have any confusions about 
51        //  how the color can be set, then here is a brief description.
52        //  First, you must be familiar with RGBA. The A is alpha and is
53        //  used for transparency. The values range from 0.0 to 1.0
54        //  To display yellow (255, 255, 0), then (255/255.0, 255/255.0, 0, 0) = 
55        //  (1.0, 1.0, 0.0, 0.0) is the argument you specify to glClearColor.
56        glClearColor (0.0, 0.0, 0.0, 0.0);
57       }
58
59    void display ()
60       {
61        //  Clear the window or more specifically the frame buffer...
62        //  This happens by replacing all the contents of the frame
63        //  buffer by the clear color (black in our case)
64        glClear (GL_COLOR_BUFFER_BIT);
65
66        /* There is nothing to be displayed */
67
68        //  Copy the contents of the frame buffer to the window so that
69        //  they can be displayed... In other words, it forces execution 
70        //  of GL commands in finite time
71        glFlush ();
72       }
73
74    void centerOnScreen (void)
75       {
76        //  Set the window position such that the window becomes centered
77        window_X_position = (glutGet (GLUT_SCREEN_WIDTH) - window_width)/2;
78        window_Y_position = (glutGet (GLUT_SCREEN_HEIGHT) - window_height)/2;
79       }
80
81    void main (int argc, char **argv)
82       {
83        //  Center the window on the screen... Notice that I'm really 
84        //  doing nothing but changing the varaibles window_X_position
85        //  and window_Y_position.
86        centerOnScreen ();
87
88        //  Connect to the windowing system + create a window
89        //  with the specified dimensions and position
90        //  + set the display mode + specify the window title.
91        glutInit(&argc, argv);
92        glutInitWindowSize (window_width, window_height);
93        glutInitWindowPosition (window_X_position, window_Y_position);
94        glutInitDisplayMode (GLUT_RGB);
95        glutCreateWindow (window_title);
96
97        //  Initialize stuff that remain constant throughout the
98        //  program.
99               init();
100
101       if (fullScreen)
102           glutFullScreen ();
103
104       // Set the display function
105       glutDisplayFunc (display);
106
107       glutMainLoop();
108     }
