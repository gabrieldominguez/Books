Introduction

This program teaches you how to open multiple sub windows inside a window. You can also use this program as a template in case you want to use sub windows in your OpenGL application.

There will be two examples on sub windows in order to make you more familiar with them...

Using the code

Example 1

1     //*************************************************************************
2     //
3     //  File Name: SubWindow Template
4     //  Author   : Ali Baderedine
5     //  
6     //  Description: SubWindow Template
7     //  
8     //*************************************************************************
9
10    #include <gl/glut.h>
11
12    // Link the lib files to the program. This is not necessary 
13    // if you have added the lib names using Project/Settings/Link
14    #pragma comment (lib, "opengl32.lib")
15    #pragma comment (lib, "glut32.lib")
16    #pragma comment (lib, "glu32.lib")
17
18    // ***** Main Window Variables *****//
19
20    #define GAP  25             /* gap between subwindows */
21
22    //  define the window position on screen
23    float main_window_x;
24    float main_window_y;
25
26    //  variables representing the window size
27    float main_window_w = 256 + GAP * 2;
28    float main_window_h = 256 + 64 + GAP * 3;
29
30    //  variable representing the window title
31    char *window_title = "SubWindow Template";
32
33    // ***** Sub Window 1 Variables *****//
34
35    //  define the window position on screen
36    float subwindow1_x;
37    float subwindow1_y;
38
39    //  variables representing the window size
40    float subwindow1_w = 256;
41    float subwindow1_h = 256;
42
43    // ***** Sub Window 2 Variables *****//
44
45    //  define the window position on screen
46    float subwindow2_x;
47    float subwindow2_y;
48
49    //  variables representing the window size
50    float subwindow2_w = 256;
51    float subwindow2_h = 64;
52
53    // ***** Main Window callback functions *****//
54
55    void main_display (void);
56    void main_reshape (int w, int h);
57
58    // ***** SubWindow 1 callback functions *****//
59
60    void subwindow1_display (void);
61
62    // ***** SubWindow 2 callback functions *****//
63
64    void subwindow2_display (void);
65
66    //  Redisplay all
67    void redisplay_all (void);
68
69    //  A program with two subwindows
70    int main_window, subwindow_1, subwindow_2;
71
72    //  Tells whether to display the window full screen or not
73    int fullScreen = 0;
74
75    //  This function centers your window on the screen
76    void centerOnScreen (void);
77
78    // ***** Main Window callback functions *****//
79
80    void main_reshape (int width,  int height)
81    {
82        //  Just take the case when the user tries
83        //  to make the size of the window very small...
84        if (width < GAP * 4 || height < GAP * 6)
85        {
86            glutSetWindow (main_window);
87            glutReshapeWindow (main_window_w, main_window_h);
88            return;
89        }
90
91        glViewport(0, 0, width, height);
92
93        //  Change the subwindow dimensions as window dimensions change
94        //  Note (Keep up the ratio!)
95
96        //  main_window_w          ---> subwindow1_w
97        //  main_window_w' (width) ---> ??
98        //  ==> 
99        subwindow1_w = (subwindow1_w * (width-GAP*2.0))/(main_window_w-GAP*2.0);
100       subwindow1_h = (subwindow1_h * (height-GAP*3.0))/(main_window_h-GAP*3.0);
101
102       glutSetWindow (subwindow_1);
103       glutPositionWindow (GAP, GAP);
104       glutReshapeWindow (subwindow1_w, subwindow1_h);
105
106       subwindow2_w = (subwindow2_w * (width-GAP*2.0))/(main_window_w-GAP*2.0);
107       subwindow2_h = (subwindow2_h * (height-GAP*3.0))/(main_window_h-GAP*3.0);
108
109       glutSetWindow (subwindow_2);
110       glutPositionWindow (GAP, GAP+subwindow1_h+GAP);
111       glutReshapeWindow (subwindow2_w, subwindow2_h);
112
113       main_window_w = width;
114       main_window_h = height;
115   }
116
117   void main_display (void)
118   {
119       glClearColor(0.0, 0.0, 0.0, 0.0);
120       glClear(GL_COLOR_BUFFER_BIT);
121
122       glutSwapBuffers();
123   }
124
125   // ***** SubWindow 1 callback functions *****//
126
127   void subwindow1_display (void)
128   {
129       glClearColor(0.0, 0.0, 1.0, 0.0);
130       glClear(GL_COLOR_BUFFER_BIT);
131
132       glColor3f (1, 1, 0);
133
134       glutWireTeapot (0.5);
135
136       glutSwapBuffers ();
137   }
138
139   // ***** SubWindow 2 callback functions *****//
140
141   void subwindow2_display (void)
142   {
143       glClearColor(1.0, 1.0, 0.0, 0.0);
144       glClear(GL_COLOR_BUFFER_BIT);
145
146       glColor3f (0, 0, 1);
147
148       glutWireTeapot (0.5);
149
150       glutSwapBuffers ();
151   }
152
153   //  Redisplay All 
154   void redisplay_all (void)
155   {
156       glutSetWindow(subwindow_1);
157       glutPostRedisplay();
158       glutSetWindow(subwindow_2);
159       glutPostRedisplay();
160   }
161
162   void centerOnScreen (void)
163   {
164       //  Set the window position such that the window becomes centered
165       main_window_x = (glutGet (GLUT_SCREEN_WIDTH) - main_window_w)/2;
166       main_window_y = (glutGet (GLUT_SCREEN_HEIGHT) - main_window_h)/2;
167   }
168
169   void main (int argc, char **argv)
170   {
171       centerOnScreen ();
172
173       /**** Main Window *****/
174       glutInit(&argc, argv);
175       glutInitWindowSize (main_window_w, main_window_h);
176       glutInitWindowPosition (main_window_x, main_window_y);
177       glutInitDisplayMode (GLUT_RGB | GLUT_DOUBLE);
178       main_window = glutCreateWindow (window_title);
179
180       glutReshapeFunc (main_reshape);
181       glutDisplayFunc (main_display);
182
183       // Subwindow 1 (Notice that x, y, w, h are just dummy values here
184       // The reason is that they're gonna be reset in main_reshape
185       // since the reshape function is called as the program starts.
186       subwindow_1 = glutCreateSubWindow (main_window, 0, 0, 0, 0);
187       glutDisplayFunc (subwindow1_display);
188
189       /**** Subwindow 2 *****/
190       subwindow_2 = glutCreateSubWindow (main_window, 0,0 , 0, 0);
191       glutDisplayFunc (subwindow2_display);
192
193       if (fullScreen)
194           glutFullScreen ();
195
196       glutMainLoop();
197   }

51    float subwindow4_w = 200;

54    float subwindow5_y;
55    float subwindow5_w = 200;
56    float subwindow5_h = 200;
57    float subwindow6_x;
58    float subwindow6_y;
59    float subwindow6_w = 200;
60    float subwindow6_h = 200;
61
62    // ***** Main Window callback functions *****//
63
64    void main_display (void);
65    void main_reshape (int w, int h);
66
67    void clear (void);
68
69    //  A program with two subwindows
70    int main_window, subwindow_1, subwindow_2, subwindow_3, 
                         subwindow_4, subwindow_5, subwindow_6;
71
72    //  This function centers your window on the screen
73    void centerOnScreen (void);
74
75    // ***** Main Window callback functions *****//
76
77    void main_reshape (int width,  int height)
78    {
79        if (width != main_window_w || height != main_window_h)
80            glutReshapeWindow (main_window_w, main_window_h);
81    }
82
83    void main_display (void)
84    {
85        glClearColor(0.0, 0.0, 0.0, 0.0);
86        glClear(GL_COLOR_BUFFER_BIT);
87
88        glutSwapBuffers();
89    }
90
91    void clear (void)
92    {
93        glClearColor (1, 1, 1, 1);
94        glClear (GL_COLOR_BUFFER_BIT);
95
96        glutSwapBuffers ();
97    }
98
99    void centerOnScreen (void)
100   {
101       //  Set the window position such that the window becomes centered
102       main_window_x = (glutGet (GLUT_SCREEN_WIDTH) - main_window_w)/2;
103       main_window_y = (glutGet (GLUT_SCREEN_HEIGHT) - main_window_h)/2;
104   }
105
106   void main (int argc, char **argv)
107   {
108       centerOnScreen ();
109
110       /**** Main Window *****/
111       glutInit(&argc, argv);
112       glutInitWindowSize (main_window_w, main_window_h);
113       glutInitWindowPosition (main_window_x, main_window_y);
114       glutInitDisplayMode (GLUT_RGB | GLUT_DOUBLE);
115       main_window = glutCreateWindow (window_title);
116
117       glutReshapeFunc (main_reshape);
118       glutDisplayFunc (main_display);
119
120       //  Set the position and dimensions of subwindows!
121       subwindow_1 = glutCreateSubWindow (main_window, GAP, GAP, 200, 200);
122       glutDisplayFunc (clear);
123       subwindow_2 = glutCreateSubWindow (main_window, 
                        200 + GAP*2, GAP, 200, 200);
124       glutDisplayFunc (clear);
125       subwindow_3 = glutCreateSubWindow (main_window, 
                        400 + GAP*3, GAP, 200, 200);
126       glutDisplayFunc (clear);
127       subwindow_4 = glutCreateSubWindow (main_window, 
                        GAP, 200 + GAP * 2, 200, 200);
128       glutDisplayFunc (clear);
129       subwindow_5 = glutCreateSubWindow (main_window, 
                        200 + GAP*2, 200 + GAP * 2, 200, 200);
130       glutDisplayFunc (clear);
131       subwindow_6 = glutCreateSubWindow (main_window, 400 + 
                        GAP*3, 200 + GAP * 2, 200, 200);
132       glutDisplayFunc (clear);
133
134
135       glutMainLoop();
136   }
