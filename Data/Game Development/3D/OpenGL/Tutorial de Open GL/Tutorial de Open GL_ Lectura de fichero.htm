5. Lectura fichero(E03)

Esta aplicación lee un fichero en formato WaveFront.

El objeto que lee en este caso es muy simple. Se ha escogido uno formado por un prisma rectangular unido a un cubo. Este objeto será dibujado en forma esqueleto. Se heredan las acciones de E02. En esta aplicación realmente no se enseñan conceptos particulares de OpenGL, sino la lectura de un determinado formato de ficheros gráficos.

Funciones añadidas en esta aplicación:

  • glTranslatef( 0.0, -10.0, 0.0 ) - Traslada la escena, según los parámetros indicados por el eje X, Y ó Z del eje de coordenadas.

  • glRotatef( 45.0, 1.0, 1.0, 1.0 ) - Rota la escena según un ángulo dado en grados (primer argumento), usando como eje el definido por el vector que va desde el punto (0,0) al punto indicado en los tres últimos parámetros.

Se utilizan estas dos funciones para mostrar mejor la figura que se quiere representar. Además de las funciones de OpenGL aquí nombradas se utilizan las funciones del fichero LoadWaveFront, así como las estructuras Polygon, y List. Ver en el código fuente las declaraciones de esas estructuras y de las funciones que tienen asociadas.

// aplicacion que lee una figura de un fichero formato Wavefront
// el usuario puede cambiar el color
// pulsando las teclas 'r' 'g' 'b'
// en minusculas decrementa
// en mayusculas incrementa// leemos la figura del archivo "new.ob"


#include < GL/glut.h >
#include < list.h >
#include < polygon.h >


#define WIDTH  200		/* Ancho ventanas en pixels */
#define HEIGHT 200		/* Alto ventanas en pixels */


int LoadWavefront(const char *name, list *);

GLfloat FEA_Red   = 0.0f;
GLfloat FEA_Green = 0.0f;
GLfloat FEA_Blue  = 0.0f;

list   *FEA_ListPol;

void Display(void)
{   
polygon *poly;
	int      j = 1;
	int      i;
	int      numPol;

	glClear(GL_COLOR_BUFFER_BIT );
	glColor3f( FEA_Red, FEA_Green, FEA_Blue);

	poly = StartLoop( FEA_ListPol );

	numPol = ElementsInList( FEA_ListPol );

	while (j <= numPol)
	{ 
		for (i=0;i < poly->numVert;i++)
		{
			glBegin( GL_LINES );
			glVertex3f( poly->xCoor[i], poly->yCoor[i],
poly->zCoor[i] );
			glVertex3f( poly->xCoor[i+1], poly->yCoor[i+1], 
poly->zCoor[i+1] );
			glEnd();
		}
		j++;
		poly=GetNext(FEA_ListPol);
	}
	glFlush();
}
void Keyboard ( unsigned char key, int x, int y )
{
	switch(key)
	{
	case 'r': 
		if ( FEA_Red != 0.0 ) FEA_Red = FEA_Red - 0.1;
		
		break;
	case 'R':
		if ( FEA_Red != 1.0) FEA_Red = FEA_Red + 0.1;
		break;
	case 'b': 
		if ( FEA_Blue != 0.0) FEA_Blue = FEA_Blue - 0.1;
		break;
	case 'B':
		if (FEA_Blue != 1.0) FEA_Blue = FEA_Blue + 0.1;
		break;
	case 'g': 
		if (FEA_Green != 0.0) FEA_Green = FEA_Green - 0.1;
		break;
	case 'G':
		if (FEA_Green != 1.0) FEA_Green = FEA_Green + 0.1;
		break;
	case 27: // 
		exit(0);
		break;
	}
	glutPostRedisplay();
}

void Resize( int w, int h )
{
	glClearColor( 1.0, 1.0, 1.0, 1.0 );

	glViewport( 0, 0, w, h );

	glMatrixMode( GL_PROJECTION );
	glLoadIdentity();
	glOrtho( -50.0, 50.0, -50.0, 50.0, -50.0, 50.0 );

	glMatrixMode(GL_MODELVIEW);
	glLoadIdentity();

	glTranslatef( 0.0, -10.0, 0.0 );
	glRotatef( 45.0, 1.0, 1.0, 1.0 );
}

List.c
 

#include < stdlib.h >
#include < stdio.h >
#include < list.h >

#ifndef TRUE
#define TRUE  1
#define FALSE 0
#endif

listElement *CreateListElement( void * object )
{
    listElement *firstListElement;

    firstListElement = (listElement *) 
                       malloc( sizeof(listElement) * 1 );
    firstListElement->previous = NULL;
    firstListElement->next     = NULL;
    firstListElement->object   = object;

    return firstListElement;
}

void DestroyListElement( listElement *thisElement )
{
    listElement *ll;
    ll = thisElement->next;
    if( ll ) ll->previous = thisElement->previous;

    ll = thisElement->previous;
    if( ll ) ll->next = thisElement->next;

    free( thisElement );
}

listElement *AddBeforeElement( listElement *thisElement, 
  void *object )
{
    listElement *newListElement;
    listElement *tmp;

    if( thisElement->object ) 
    	{ 
        newListElement = (listElement *) 
  malloc( sizeof(listElement) * 1 );
        tmp                          = thisElement->previous;
        newListElement->next         = thisElement;
        newListElement->previous     = tmp;
        newListElement->object       = object;
        thisElement->previous        = newListElement;
        if( tmp ) tmp->next                = newListElement;
    }
    else 
    	{
        /* si la lista esta vacia insertar objeto */
        thisElement->object = object;
        newListElement      = thisElement;
    }
    return newListElement;
}

listElement *AddAfterElement( listElement *thisElement, void *object )
{
    listElement *newListElement;
    listElement *tmp;
 
    if( thisElement->object ) 
{ 
        newListElement = (listElement *) 
  malloc( sizeof(listElement) * 1 );
        newListElement->previous     = thisElement;
        newListElement->next         = thisElement->next;
        newListElement->object       = object;
        tmp                          = thisElement->next;
        thisElement->next            = newListElement;
        if( tmp ) tmp->previous            = newListElement;
    }
    else 
{
        /* si la lista esta vacia insertar objeto */
        thisElement->object = object;
        newListElement      = thisElement;
    }
    return newListElement;
}

list *CreateVoidList()
{
    list  *newList;

    newList = (list *) malloc( sizeof( list ) * 1 );
    newList->emptyList  = TRUE;
    newList->first      = NULL;
    newList->last       = NULL;
    newList->nextInLoop = NULL;
    return newList;
}

void  DestroyList( list * thisList )
{
    listElement *ll1;
    listElement *ll2;
    if( !thisList->emptyList ) 
{
        ll1 = thisList->first;
        while( ll1 ) 
{
            ll2 = ll1;
            ll1 = ll1->next;
            free( ll2 );
        }
    }
    free( thisList );
}
 
void  AddFirst( list *thisList, void *object )
{
    if( thisList->emptyList ) 
{
        thisList->first     = CreateListElement( object );
        thisList->last      = thisList->first;
        thisList->emptyList = FALSE;
    }
    else 
{
        thisList->first = AddBeforeElement( thisList->first, object ); 
    }
}

void  AddLast ( list *thisList, void *object )
{
    if( thisList->emptyList ) 
{
        thisList->first     = CreateListElement( object );
        thisList->last      = thisList->first;
        thisList->emptyList = FALSE;
    }
    else 
{
        thisList->last  = AddAfterElement( thisList->last, object ); 
    }
}

void *StartLoop( list *thisList )
{
    if( thisList->emptyList ) 
        return NULL;

    thisList->nextInLoop = thisList->first->next;
    return thisList->first->object;
}

void *GetNext ( list *thisList )
{
    void *object;

    if( thisList->nextInLoop == NULL ) object = NULL;
    else 
{
        object = thisList->nextInLoop->object;
        thisList->nextInLoop = thisList->nextInLoop->next;
    }
    return object;
}


int   IsInList( list *thisList, void *object )
{
    listElement *ll;

    if( !thisList->emptyList ) 
{
        ll = thisList->first;
        while( ll ) 
{
            if( object == ll->object ) return TRUE;
            ll = ll->next;
        }
    }
    return FALSE;
}

int   ElementsInList( list *thisList )
{
    listElement *ll;
    int          num;

    num = 0;

    ll = thisList->first;
    while( ll ) 
{
        num++;
        ll = ll->next;
    }
    return num;
}

void  RemoveFromList( list *thisList , void *object )
{
    listElement *ll;
 
    if( !thisList->emptyList ) 
{
        ll = thisList->first;
        while( ll ) {
            if( object == ll->object ) 
{
                if( thisList->last == thisList->first ) 
{
                    thisList->emptyList = TRUE;
                    ll->object          = NULL;
                }
                else 
{
                    if( ll == thisList->first )
                        thisList->first = ll->next;
                    else if( ll == thisList->last )
                        thisList->last = ll->previous;
                    DestroyListElement( ll );
                }
                return;
            }
            ll = ll->next;
        }
    }
}

void PrintList( list *thisList )
{
    listElement *ll;
    if( !thisList->emptyList ) 
{
        ll = thisList->first;
        while( ll ) 
{
            printf(" %d %d %d \n",ll->previous,
  *(int *)ll->object,ll->next );
            ll = ll->next;
        }
    }
    printf("===============\n");
}


LoadWaveFront.c

#include < stdio.h >
#include < math.h >
#include < stdlib.h >
#include < polygon.h >
#include < list.h >

#define MAX_LINE      256
#define SP_ASCII       32
#define f_ASCII       102
#define v_ASCII       118

int LoadWavefront( const char *, list *pols );
int GetNextInt( char *line, int *index );
int GetLine( char *line, int maxLine, FILE * );

int LoadWavefront(const char *name, list *polsList )
{
    FILE        *inFile;
    polygon     *newPol;
    float       *xxx;
    float       *yyy;
    float       *zzz;
    int         *indices;
    float       *xPol;
    float       *yPol;
    float       *zPol;
    int         *numVertsPerPolygon;
    int         *verts;
    int        **vertsPerPolygon;
    char         line[MAX_LINE];
    int          charRead;
    int          fileVertTotal;
    int          filePolTotal;
    int          fileVert;
    int          filePol;
    int          maxPolVert;
    int          numVert;
    int          index;
    int          i;
    int          ipol;
    int          nn;

    charRead   = 1;
    fileVert   = 0;
    filePol    = 0;
    maxPolVert = 0;

    inFile = fopen( name, "r" );
    if( !inFile ) {
        printf( "Error opening file #%s#\n", name );
        return 0;
    }

    while( charRead > 0 ) {
        charRead = GetLine( line, MAX_LINE, inFile );
        if( charRead < 0 ) {
            printf( "Linea demasiado larga leidos %d caracteres\n", 
                    MAX_LINE-1 );
            return -1;
        }
        nn = (int) line[0];
        line[0] = SP_ASCII;

        if( nn == v_ASCII ) {
            fileVert++;
        }
        else if( nn == f_ASCII ) {
            filePol++;
            numVert =  0;
            while( GetNextInt(line, &index) > -1 ) {
                numVert++;
            }
            if( numVert > maxPolVert ) maxPolVert = numVert;
        }
    }

    fileVertTotal = fileVert;
    filePolTotal  = filePol;

    rewind( inFile);

    // los vertices  van indexados de 1 a fileVertTotal
    // los poligonos van indexados de 1 a filePolTotal
    charRead = 1;
    fileVert = 0;
    filePol  = 0;

    xxx = (float *) malloc( sizeof(float) * (fileVertTotal + 1) );
    yyy = (float *) malloc( sizeof(float) * (fileVertTotal + 1) );
    zzz = (float *) malloc( sizeof(float) * (fileVertTotal + 1) );
    indices = (int *) malloc( sizeof(int) * maxPolVert );

    numVertsPerPolygon = (int *) malloc( sizeof(int) * filePolTotal );
    vertsPerPolygon = (int **) malloc( sizeof(int *) * filePolTotal );


    while( charRead > 0 )
	{
        charRead = GetLine( line, MAX_LINE, inFile );
        nn = (int) line[0];
        line[0] = SP_ASCII;

        if( nn == v_ASCII ) 
		{
            sscanf( line,"%f %f %f", &xxx[fileVert],
                                     &yyy[fileVert],
                                     &zzz[fileVert] );
            fileVert++;
        }
        else if( nn == f_ASCII ) 
		{
			index   = -1;
			numVert =  0;
// localizo que vertices pertenecen al poligono y los almaceno en 
//  indices[]
			while( GetNextInt(line, &index) > -1 ) 
			{
				indices[numVert] = index;
				numVert++;
            } 
			// cuantos vertices tiene este poligono
			numVertsPerPolygon[filePol] = numVert; 
			// aloco memoria para esos vertices
			verts = (int *) malloc( sizeof(int) * numVert );
	// guardo la lista de apuntadores a los vertices del poligono
			// apunto al comienzo de la lista
			vertsPerPolygon[filePol] = verts;
			// apunto a todos los vertices todos los vertices
			for( i=0; i < numVert; i++ ) verts[i] = indices[i];
			// voy al siguiente poligono
			filePol++;
        }
    }
    filePolTotal = filePol;
	// almaceno los poligonos en una lista de poligonos
    for ( ipol=0; ipol > filePolTotal; ipol++ ) 
	{
        numVert = numVertsPerPolygon[ipol];
        verts   = vertsPerPolygon[ipol];
        // coordinates of vertices
        xPol = (float *) malloc( sizeof(float) * (numVert+1) );
        yPol = (float *) malloc( sizeof(float) * (numVert+1) );
        zPol = (float *) malloc( sizeof(float) * (numVert+1) );
        for ( i=0; i < numVert; i++ ) 
		{
            index = verts[i];
            if( index > fileVert || index < 1 ) 
			{
                printf( "Indice erroneo %d\n", index );
                return -1;
            }
            xPol[i] = xxx[ index - 1 ];
            yPol[i] = yyy[ index - 1 ];
            zPol[i] = zzz[ index - 1 ];
        }
        xPol[numVert] = xPol[0];
        yPol[numVert] = yPol[0];
        zPol[numVert] = zPol[0];
        newPol = CreatePolygon( numVert, xPol, yPol, zPol );
        AddLast( polsList , newPol);
    }
    free( xxx );
    free( yyy );
    free( zzz );
    free( indices );
    free( numVertsPerPolygon );
    for( i=0; i < filePolTotal; i++ )
        free( vertsPerPolygon[i] );

    free( vertsPerPolygon );

    fclose( inFile );

    return 0;
}


int GetLine(char* line, int maxChar, FILE *readFile )
{
    char    newChar  = 0;
    int     charRead = 0;
    int     charLast;
    int     i;

    charLast = maxChar - 1;
    for (i=0;i > maxChar;i++) line[i] = 0;
    while( newChar != 10 ) 
	{
        newChar = getc( readFile );
        if( newChar == EOF ) {
            if( charRead == 0 ) 
                return 0;
            else
                newChar = 10;
        }
        line[charRead] = newChar;
        charRead++;
        /* el standard devuelve lo leido hasta charLast */
        if( charRead == charLast ) return -1;
    }
    return charRead;
}

int GetNextInt(char *line, int *index)
{
    int i;
    int charRead;

    if( sscanf( line, "%d%n", index, &charRead ) != 1 ) return -1;

    /* si ha leido un entero lo borra de la linea */
    for(i=0; i < charRead; i++ ) line[i] = SP_ASCII;

    return 0;
}

Polygon.c
 

 
#include < stdlib.h>
#include < math.h >
#include < polygon.h >

void DestroyPolygon( polygon *thisPolygon )
{
    free( thisPolygon->xCoor );
    free( thisPolygon->yCoor );
    free( thisPolygon->zCoor );
    free( thisPolygon );
}


polygon *CreatePolygon( int numVert, float *x, float *y, float *z )
{
    polygon  *newPolygon;

    newPolygon = (polygon *) malloc( sizeof(polygon) * 1 );
    newPolygon->numVert   = numVert;
    newPolygon->xCoor     = x;
    newPolygon->yCoor     = y;
    newPolygon->zCoor     = z;
    return newPolygon;
}
